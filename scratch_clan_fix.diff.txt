+func _step(step: int) -> bool:
+	var ids := offered_ids()
+	if ids.is_empty():
+		return false
+	var index := ids.find(_selected_clan)
+	# Nothing is picked: `down` lands on the FIRST entry and `up` on the LAST, which is
+	# what those two lines mean. Seeding the index and THEN adding `step` would skip an
+	# entry at both ends — `SectScreen._step` does exactly that (`index = 0 if step > 0`

+	# changed there: `sect_screen.gd` is not this slice's file.

+	_selected_clan = String(ids[next])

+	return true

+## `SectScreen.on_stack_input` gives: a screen that declines an unbound or cancelled

+func on_stack_input(event: InputEvent) -> bool:
+	_bind_nodes()
