+## The clan ids this page is offering to enter, in canonical order. The CATALOG, read
+## off the facade's own `summary()["clans"]` — the same dict [method select_clan]
+## validates against and the same one `act_join`'s pick is drawn from, so what a player
+## can cycle through is exactly what the page can name. A member is offered the whole
+## catalog too, because leaving and rejoining are two acts in order and the roster a
+## member needs is the roster they left.
+func offered_ids() -> Array:
+	_bind_nodes()
+	var clans := _codex.get("clans", {}) as Dictionary
+	var ids: Array[String] = []
+	for key in clans.keys():
+		ids.append(String(key))
+	ids.sort()
+	return ids

+## Move the pick `step` entries along the offered list, wrapping. Declines (returns
+## false) when there is nothing to pick, so `ui_down` on an empty catalog is not
+## consumed and the player's `ui_cancel` is left for `ScreenStack`.
+##
+## **This is what makes the join pressable by a PLAYER rather than only by a test.**
+## `act_join` re-runs the module's gate and is the sole production caller of
+## `ClanApi.join`, but it acts on `_selected_clan`, and before this method NOTHING in
+## `game/src` ever called [method select_clan] — no panel, no widget, no key. So
+## `can_join()` was permanently false, the `join` control was permanently disabled, and
+## the only way to reach the membership every authored `household_heir_registered` quest
+## needs was a test calling `select_clan` itself. The verb existed; the gesture that
+## arms it did not.
+##
+## ## Why a key cycle and not a roster widget
+##
+## This page has no roster scene of its own, and one it grew would be a SECOND house
+## list beside the catalog the facade already publishes — the exact duplication
+## [method offered_ids] refuses. `SectScreen` publishes `offered_ids` + `_step` +
+## `on_stack_input` for exactly this page's sibling problem, and this is that same
+## three-piece shape rather than a new one.
+func _step(step: int) -> bool:
+	var ids := offered_ids()
+	if ids.is_empty():
+		return false
+	var index := ids.find(_selected_clan)
+	# Nothing is picked: `down` lands on the FIRST entry and `up` on the LAST, which is
+	# what those two lines mean. Seeding the index and THEN adding `step` would skip an
+	# entry at both ends — `SectScreen._step` does exactly that (`index = 0 if step > 0`
+	# then `posmod(index + step, …)` makes the first `down` land on the second house), so
+	# the two pages would disagree about where the first keypress goes. Noted rather than
+	# changed there: `sect_screen.gd` is not this slice's file.
+	var next := 0 if step > 0 else ids.size() - 1
+	if index >= 0:
+		next = posmod(index + step, ids.size())
+	_selected_clan = String(ids[next])
+	_render()
+	return true
+

+## `ui_accept` runs the page's primary verb; `ui_up` / `ui_down` walk the pick.
+##
+## The guards are merged into one clause on purpose rather than stacked, for the reason
+## `SectScreen.on_stack_input` gives: a screen that declines an unbound or cancelled
+## event by DECLARING it declined is exactly the contract `ScreenStack` relies on, and
+## one decision says it once. `ui_cancel` is excluded so the stack still pops.
+func on_stack_input(event: InputEvent) -> bool:
+	_bind_nodes()
+	if _actor == null or event == null or not _bound:
+		return false
+	if not event.is_pressed() or event.is_echo() or event.is_action_pressed(&"ui_cancel"):
+		return false
+	if event.is_action_pressed(&"ui_accept"):
+		return _accept()
+	if event.is_action_pressed(&"ui_down"):
+		return _step(1)
+	if event.is_action_pressed(&"ui_up"):
+		return _step(-1)
+	return false
